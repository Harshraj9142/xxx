import pandas as pd
import numpy as np

# Read CSV file
data = pd.read_csv("data.csv")

# Input (X) and Output (Y)
X = data["Class_Test"].values
Y = data["Semester"].values

# ------------------------------------------------
# FORMULA FOR LINEAR REGRESSION
#
# y = mx + c
#
# m = Σ(x - x_mean)(y - y_mean)
#     -----------------------------
#       Σ(x - x_mean)^2
#
# c = y_mean - m * x_mean
# ------------------------------------------------

# Calculate mean
x_mean = np.mean(X)
y_mean = np.mean(Y)

# Calculate slope (m)
m = np.sum((X - x_mean) * (Y - y_mean)) / np.sum((X - x_mean) ** 2)

# Calculate intercept (c)
c = y_mean - m * x_mean

# Predicted values
Y_pred = m * X + c

# ------------------------------------------------
# MAE FORMULA
#
# MAE = (1/n) * Σ |y - y_pred|
# ------------------------------------------------

MAE = np.mean(np.abs(Y - Y_pred))

# ------------------------------------------------
# MSE FORMULA
#
# MSE = (1/n) * Σ (y - y_pred)^2
# ------------------------------------------------

MSE = np.mean((Y - Y_pred) ** 2)

# ------------------------------------------------
# RMSE FORMULA
#
# RMSE = √MSE
# ------------------------------------------------

RMSE = np.sqrt(MSE)

# ------------------------------------------------
# R² FORMULA
#
# R² = 1 - [Σ(y - y_pred)^2 / Σ(y - y_mean)^2]
# ------------------------------------------------

R2 = 1 - (
    np.sum((Y - Y_pred) ** 2) /
    np.sum((Y - y_mean) ** 2)
)

# ------------------------------------------------
# Predict semester marks for class test = 20
# ------------------------------------------------

x_new = 20
y_new = m * x_new + c

# Display results
print("Slope (m):", m)
print("Intercept (c):", c)

print("\nLinear Regression Equation:")
print("y =", m, "x +", c)

print("\nPredicted Semester Marks for Class Test = 20:")
print(y_new)

print("\nError Metrics:")
print("MAE  =", MAE)
print("MSE  =", MSE)
print("RMSE =", RMSE)
print("R2   =", R2)
