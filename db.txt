import pandas as pd
import numpy as np

# -----------------------------------------
# Read CSV file
# -----------------------------------------

data = pd.read_csv("data.csv")

points = data["Point"].values
X = data[["X", "Y"]].values.astype(float)

# DBSCAN parameters
epsilon = 2
min_pts = 2


# -----------------------------------------
# Euclidean Distance Formula
#
# d = sqrt((x1-x2)^2 + (y1-y2)^2)
# -----------------------------------------

def distance(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))


# -----------------------------------------
# Find points inside epsilon neighborhood
# -----------------------------------------

def region_query(X, point_index, epsilon):

    neighbors = []

    for i in range(len(X)):

        if distance(X[point_index], X[i]) <= epsilon:
            neighbors.append(i)

    return neighbors


# -----------------------------------------
# DBSCAN Algorithm
# -----------------------------------------

def dbscan(X, epsilon, min_pts):

    n = len(X)

    # Initially all points are unvisited
    visited = [False] * n

    # -1 means noise
    cluster_labels = [-1] * n

    cluster_id = 0

    for i in range(n):

        # Skip if already visited
        if visited[i]:
            continue

        # Mark point as visited
        visited[i] = True

        # Find epsilon-neighbors
        neighbors = region_query(X, i, epsilon)

        # If number of neighbors < MinPts,
        # mark as noise
        if len(neighbors) < min_pts:

            cluster_labels[i] = -1

        else:

            # Create new cluster
            cluster_id += 1

            cluster_labels[i] = cluster_id

            # Expand the cluster
            j = 0

            while j < len(neighbors):

                neighbor = neighbors[j]

                # If neighbor is not visited
                if not visited[neighbor]:

                    visited[neighbor] = True

                    # Find its neighbors
                    new_neighbors = region_query(
                        X,
                        neighbor,
                        epsilon
                    )

                    # If it is a core point,
                    # add its neighbors
                    if len(new_neighbors) >= min_pts:

                        for p in new_neighbors:

                            if p not in neighbors:
                                neighbors.append(p)

                # Assign point to cluster
                if cluster_labels[neighbor] == -1:
                    cluster_labels[neighbor] = cluster_id

                j += 1

    return cluster_labels


# -----------------------------------------
# Apply DBSCAN
# -----------------------------------------

labels = dbscan(
    X,
    epsilon,
    min_pts
)


# -----------------------------------------
# Display Results
# -----------------------------------------

print("DBSCAN Results")
print("--------------")

for i in range(len(points)):

    if labels[i] == -1:
        print(points[i], "-> Noise")
    else:
        print(
            points[i],
            "-> Cluster",
            labels[i]
        )
